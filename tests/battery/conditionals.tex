\def\yn#1{\message{<#1>}}
\ifnum 1<2 \yn{ifnum-lt}\else\yn{BAD}\fi
\ifnum 2<1 \yn{BAD}\else\yn{ifnum-else}\fi
\ifnum \count5=7 \yn{ifnum-reg}\else\yn{BAD}\fi
\ifdim 1pt>0pt \yn{ifdim-gt}\else\yn{BAD}\fi
\ifdim \dimen5=3pt \yn{ifdim-reg}\else\yn{BAD}\fi
\ifodd 3 \yn{ifodd}\else\yn{BAD}\fi
\ifodd 4 \yn{BAD}\else\yn{ifodd-else}\fi
\ifvmode \yn{ifvmode}\else\yn{BAD}\fi
\ifhmode \yn{BAD}\else\yn{ifhmode-else}\fi

\ifinner \yn{BAD}\else\yn{ifinner-else}\fi
\ifvoid 9 \yn{ifvoid}\else\yn{BAD}\fi
\ifhbox 5 \yn{ifhbox}\else\yn{BAD}\fi
\ifvbox 5 \yn{BAD}\else\yn{ifvbox-else}\fi
\ifx \relax\relax \yn{ifx-same}\else\yn{BAD}\fi
\ifx \relax\undefinedthing \yn{BAD}\else\yn{ifx-diff}\fi
\ifx \undefinedone\undefinedtwo \yn{ifx-both-undef}\else\yn{BAD}\fi
\if AA\yn{if-char}\else\yn{BAD}\fi
\if AB\yn{BAD}\else\yn{if-char-else}\fi

\ifcase -1 \yn{BAD}\or\yn{BAD}\else\yn{ifcase-negative}\fi
\ifcase 3 \yn{BAD}\or\yn{BAD}\fi \yn{ifcase-no-else}
\ifcase 0 \yn{ifcase-zero}\or\yn{BAD}\else\yn{BAD}\fi
