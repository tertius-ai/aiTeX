% `tex.web` §1211 keeps accumulated prefixes local to `prefixed_command`.
% Expanded blanks and `\relax` are skipped while it looks for the assignment;
% once that assignment finishes, the prefix is gone even for state whose
% global/local distinction TeX deliberately ignores.
\catcode`\{=1 \catcode`\}=2
\font\f=cmr10 \f

% A blank can arrive from expansion rather than literally after the prefix.
\count0=0
{\def\blank{ }\global\blank\count0=1}
\message{<blank=\the\count0>}

% Relax is skipped by the same scanner.
\count1=0
{\global\relax\count1=2}
\message{<relax=\the\count1>}

% These assignments accept and consume `\global`, but their values do not use
% the ordinary eqtb local/global machinery.  The following counts must remain
% local to their groups rather than inheriting a stale prefix.
\count2=0
{\global\prevdepth=1pt \count2=2}
\message{<prevdepth=\the\count2>}

\count3=0
{\global\prevgraf=0 \count3=3}
\message{<prevgraf=\the\count3>}

\count4=0
\setbox0=\hbox{\global\spacefactor=1000 \count4=4}
\message{<spacefactor=\the\count4>}

\count5=0 \setbox0=\hbox{}
{\global\wd0=1pt \count5=5}
\message{<boxdim=\the\count5><wd=\the\wd0>}

\end
