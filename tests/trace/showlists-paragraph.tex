% `tex.web` §218: the nest level `\showlists` never mentioned.
%
% §1091's `new_graf` pushes a level for the paragraph like any other, and
% §218 walks the whole nest — so a `\showlists` in the middle of an ordinary
% paragraph on the page's own vertical list prints a `### horizontal mode`
% block for it, above the `### vertical mode` block for the page.
%
% This engine keeps that paragraph outside `ts.open` — which holds the boxes —
% so it was the one level the display skipped. A `\showlists` in the middle of
% a paragraph is not an exotic place to put one, and the block it prints is
% the material a reader is usually asking about.
%
% §219's tail for a horizontal level is the language triple:
%
%     if m=hmode then
%       if pg_field<>@'40600000 then
%         begin print(" (language"); print_int(pg_field mod @'200000);
%         print(":hyphenmin"); print_int(pg_field div @'20000000);
%         print_char(","); print_int((pg_field div @'200000) mod @'100);
%         print_char(")");
%         end;
%
% — `\language`, `\lefthyphenmin` and `\righthyphenmin` as they stood when the
% paragraph began, which is what decides how the words in it may be broken.
\catcode`\{=1 \catcode`\}=2
\tracingonline=1 \showboxbreadth=99 \showboxdepth=99
\hsize=100pt \parindent=0pt \vsize=200pt
\font\r=cmr10 \r \hbadness=10000 \vbadness=10000
\showthe\escapechar

% A paragraph on the page's own vertical list.
A B\showlists\par

% ...with the triple set to something other than the default, then changed
% after the paragraph has begun.  The heading keeps 7:3,4 while the trailing
% current-language field follows the automatic language node to 8.
\language=7 \lefthyphenmin=3 \righthyphenmin=4
C \language=8 \lefthyphenmin=5 \righthyphenmin=6 D\showlists\par

% Not covered: a paragraph inside a `\vbox`. TeX has *two* levels there — the
% box and the paragraph in it — and this engine has one `OpenBox` serving as
% both, so it prints the paragraph and not the box above it. That is a nest
% shape, not a display, and it is the last thing §218 wants here.
\showthe\escapechar
\end
